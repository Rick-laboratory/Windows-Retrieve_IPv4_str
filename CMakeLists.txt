cmake_minimum_required(VERSION 3.16)
project(LocalIPv4Lookup LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This Winsock example requires Windows.")
endif()
add_executable(local-ipv4 main.cpp)
target_compile_features(local-ipv4 PRIVATE cxx_std_17)
set_target_properties(local-ipv4 PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(local-ipv4 PRIVATE _WIN32_WINNT=0x0600)
target_link_libraries(local-ipv4 PRIVATE ws2_32)
if(MSVC)
    target_compile_options(local-ipv4 PRIVATE /W4 /permissive-)
else()
    target_compile_options(local-ipv4 PRIVATE -Wall -Wextra -Wpedantic)
endif()
